# Clone
git clone <url>
cd repo

# Configure remotes
git remote -v
git remote add upstream <original-url>

# Sync main
git switch main
git fetch upstream
git merge upstream/main
git push origin main

# Create feature branch
git switch -c feature-name

# Work
git status
git diff
git add .
git commit -m "Meaningful message"

# Push
git push -u origin feature-name

# Update branch before merge
git fetch upstream
git switch main
git merge upstream/main
git switch feature-name
git merge main

# OR
git rebase main

# Resolve conflicts
git status
git add <file>

# Finish merge
git commit

# OR finish rebase
git rebase --continue

# Push updated branch
git push
# OR
git push --force-with-lease

# Cleanup
git switch main
git fetch upstream
git merge upstream/main
git push origin main
git branch -d feature-name
git push origin --delete feature-name
